# ============================================================
# İLK KURULUM (Yeni Bilgisayar - GitHub'dan)
# ============================================================

# 1. Projeyi GitHub'dan indir
cd c:\proje
git clone https://github.com/Eraysarkaya/quiz-ai.git
cd quiz-ai

# 2. Python kütüphanelerini yükle
pip install -r requirements.txt

# 3. Frontend bağımlılıklarını yükle
cd frontend
npm install
cd ..

# 4. .env dosyasını oluştur
copy .env.example .env
# Not: .env dosyasını aç ve GROQ_API_KEY değerini gir

# 5. (İlk kurulumda) Vektör veritabanı zaten mevcut
# Eğer yeniden oluşturmak istersen:
# python -m backend.scripts.load_dataset
# python -m backend.scripts.chunking
# python -m backend.scripts.build_index
